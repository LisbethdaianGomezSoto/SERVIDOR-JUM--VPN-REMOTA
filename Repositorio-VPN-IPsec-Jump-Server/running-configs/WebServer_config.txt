##### 1. RED  (Edit config del nodo en GNS3 -> /etc/network/interfaces)
auto eth0
iface eth0 inet static
  address 10.7.1.138
  netmask 255.255.255.248
  gateway 10.7.1.137
  up echo "nameserver 8.8.8.8" > /etc/resolv.conf


##### 2. VERIFICACION DE RED
# $ ip addr
#   eth0: inet 10.7.1.138/29 scope global eth0
# $ ip route
#   default via 10.7.1.137 dev eth0
#   10.7.1.136/29 dev eth0 scope link  src 10.7.1.138


##### 3. DNS  (/etc/resolv.conf)
nameserver 8.8.8.8


##### 4. SERVICIOS
# HTTPS  -> Apache2 (443)  : Sistema de Caja
# SSH    -> OpenSSH (22)
# RDP    -> xrdp (3389)    : login con el usuario rdpuser

mkdir -p /run/sshd
/usr/sbin/sshd
service xrdp start


##### 5. USUARIOS LOCALES
# rdpuser : home /home/rdpuser, shell /bin/bash (usado en Putty y xrdp)
# La contrasena no se publica en el repositorio.
useradd -m -s /bin/bash rdpuser
passwd rdpuser
